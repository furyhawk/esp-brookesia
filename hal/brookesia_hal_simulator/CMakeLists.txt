set(COMPONENT_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(COMPONENT_SRC_DIR ${COMPONENT_DIR}/src)
set(COMPONENT_INCLUDE_DIRS ${COMPONENT_DIR}/include)
set(COMPONENT_PRIVATE_INCLUDE_DIRS ${COMPONENT_SRC_DIR})

file(GLOB_RECURSE COMPONENT_SRCS_C ${COMPONENT_SRC_DIR}/*.c)
file(GLOB_RECURSE COMPONENT_SRCS_CPP ${COMPONENT_SRC_DIR}/*.cpp)

if(NOT TARGET brookesia::lib_utils)
    add_subdirectory(
        ${COMPONENT_DIR}/../../utils/brookesia_lib_utils
        ${CMAKE_BINARY_DIR}/brookesia_lib_utils
    )
endif()
if(NOT TARGET brookesia::hal_interface)
    add_subdirectory(
        ${COMPONENT_DIR}/../brookesia_hal_interface
        ${CMAKE_BINARY_DIR}/brookesia_hal_interface
    )
endif()

set(COMPONENT_LIB brookesia_hal_simulator)
add_library(${COMPONENT_LIB} STATIC
    ${COMPONENT_SRCS_C}
    ${COMPONENT_SRCS_CPP}
)
target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_20)
target_include_directories(${COMPONENT_LIB}
    PUBLIC
        ${COMPONENT_INCLUDE_DIRS}
    PRIVATE
        ${COMPONENT_PRIVATE_INCLUDE_DIRS}
)
target_link_libraries(${COMPONENT_LIB}
    PUBLIC
        brookesia::hal_interface
        brookesia::lib_utils
)

set(BROOKESIA_HAL_SIMULATOR_GENERAL_DEVICE_PLUGIN_SYMBOL general_simulator_device_symbol)
set(BROOKESIA_HAL_SIMULATOR_AUDIO_DEVICE_PLUGIN_SYMBOL audio_simulator_device_symbol)
set(BROOKESIA_HAL_SIMULATOR_DISPLAY_DEVICE_PLUGIN_SYMBOL display_simulator_device_symbol)
set(BROOKESIA_HAL_SIMULATOR_POWER_DEVICE_PLUGIN_SYMBOL power_simulator_device_symbol)
set(BROOKESIA_HAL_SIMULATOR_STORAGE_DEVICE_PLUGIN_SYMBOL storage_simulator_device_symbol)
set(BROOKESIA_HAL_SIMULATOR_WIFI_DEVICE_PLUGIN_SYMBOL wifi_simulator_device_symbol)
target_compile_definitions(${COMPONENT_LIB} PRIVATE
    CONFIG_BROOKESIA_HAL_SIMULATOR_ENABLE_GENERAL_DEVICE=1
    CONFIG_BROOKESIA_HAL_SIMULATOR_ENABLE_AUDIO_DEVICE=1
    CONFIG_BROOKESIA_HAL_SIMULATOR_ENABLE_DISPLAY_DEVICE=1
    CONFIG_BROOKESIA_HAL_SIMULATOR_ENABLE_POWER_DEVICE=1
    CONFIG_BROOKESIA_HAL_SIMULATOR_ENABLE_WIFI_DEVICE=1
    CONFIG_BROOKESIA_HAL_SIMULATOR_ENABLE_STORAGE_DEVICE=1
    BROOKESIA_HAL_SIMULATOR_GENERAL_DEVICE_PLUGIN_SYMBOL=${BROOKESIA_HAL_SIMULATOR_GENERAL_DEVICE_PLUGIN_SYMBOL}
    BROOKESIA_HAL_SIMULATOR_AUDIO_DEVICE_PLUGIN_SYMBOL=${BROOKESIA_HAL_SIMULATOR_AUDIO_DEVICE_PLUGIN_SYMBOL}
    BROOKESIA_HAL_SIMULATOR_DISPLAY_DEVICE_PLUGIN_SYMBOL=${BROOKESIA_HAL_SIMULATOR_DISPLAY_DEVICE_PLUGIN_SYMBOL}
    BROOKESIA_HAL_SIMULATOR_POWER_DEVICE_PLUGIN_SYMBOL=${BROOKESIA_HAL_SIMULATOR_POWER_DEVICE_PLUGIN_SYMBOL}
    BROOKESIA_HAL_SIMULATOR_WIFI_DEVICE_PLUGIN_SYMBOL=${BROOKESIA_HAL_SIMULATOR_WIFI_DEVICE_PLUGIN_SYMBOL}
    BROOKESIA_HAL_SIMULATOR_STORAGE_DEVICE_PLUGIN_SYMBOL=${BROOKESIA_HAL_SIMULATOR_STORAGE_DEVICE_PLUGIN_SYMBOL}
)
target_link_libraries(${COMPONENT_LIB} PRIVATE "-u ${BROOKESIA_HAL_SIMULATOR_GENERAL_DEVICE_PLUGIN_SYMBOL}")
target_link_libraries(${COMPONENT_LIB} PRIVATE "-u ${BROOKESIA_HAL_SIMULATOR_AUDIO_DEVICE_PLUGIN_SYMBOL}")
target_link_libraries(${COMPONENT_LIB} PRIVATE "-u ${BROOKESIA_HAL_SIMULATOR_DISPLAY_DEVICE_PLUGIN_SYMBOL}")
target_link_libraries(${COMPONENT_LIB} PRIVATE "-u ${BROOKESIA_HAL_SIMULATOR_POWER_DEVICE_PLUGIN_SYMBOL}")
target_link_libraries(${COMPONENT_LIB} PRIVATE "-u ${BROOKESIA_HAL_SIMULATOR_STORAGE_DEVICE_PLUGIN_SYMBOL}")
target_link_libraries(${COMPONENT_LIB} PRIVATE "-u ${BROOKESIA_HAL_SIMULATOR_WIFI_DEVICE_PLUGIN_SYMBOL}")

add_library(brookesia::hal_simulator ALIAS ${COMPONENT_LIB})
